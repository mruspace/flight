# Fault schedule for a Mru Flight run: one fault per line, kind:replica@tick.
# Copy to faults.txt next to run.sh to use it. Times are in steps (ticks).
# For long runs. The OPS-SAT package uses faults.5min.txt; it ran in orbit on 18 October 2026, and run 2's schedules on 22 October 2026 (docs/opssat.md).
#
#   kill     the replica process is terminated
#   stuck    the replica keeps returning the same wrong value
#   hang     the replica stops answering
#   corrupt  one bit flip in the replica's working memory, in its next computation
#   stuckbit one bit forced to 0 or 1 from that step on: stuckbit:I@T/LOC.BIT.V,
#            LOC is in, block, acc or out
#   sensor   one bit flip in the radiation sensor (a self-test of the sensor)
#
sensor:0@1000
corrupt:2@50000
stuck:1@200000
kill:0@600000
